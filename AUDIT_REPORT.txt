export function validateName(raw: string): ValidationResult {
  const cleaned = raw.trim().replace(/[^a-zA-Z\s'\-]/g, '').trim();
  if (cleaned.length < 2) return { valid: false };
  const words = cleaned.split(/\s+/).filter(Boolean);
  if (words.length === 0 || words.length > 4) return { valid: false };
  const validWord = /^[a-zA-Z'\-]{1,}$/;
  if (!words.every((w) => validWord.test(w))) return { valid: false };
  const noise = new Set(['uh', 'um', 'ah', 'er', 'hmm', 'yes', 'no', 'okay', 'ok', 'hi', 'hello', 'hey']);
  if (words.length === 1 && noise.has(words[0].toLowerCase())) return { valid: false };
  const normalized = words.map((w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()).join(' ');
  return { valid: true, value: normalized };
}

--- Phone Validation (server/src/lib/validators.ts:81-95) ---
export function validatePhone(raw: string): ValidationResult {
  let s = expandSpokenDigits(raw);
  s = s.replace(/[\s\-().+]/g, '');
  s = s.replace(/^91/, '');
  s = s.replace(/^0/, '');
  const match = s.match(/([6-9]\d{9})/);
  if (!match) return { valid: false };
  return { valid: true, value: match[1] };
}

--- Date Validation (server/src/lib/validators.ts:207-221) ---
export function validateDate(raw: string): ValidationResult {
  const d = parseNaturalDate(raw);
  if (!d || isNaN(d.getTime())) return { valid: false };
  const today = startOfDay(new Date());
  if (d < today) return { valid: false };
  if (!CLINIC_OPEN_DAYS.includes(d.getDay())) return { valid: false };
  return { valid: true, value: formatDate(d) };
}

--- Time Validation (server/src/lib/validators.ts:285-303) ---
export function validateTime(raw: string): ValidationResult {
  const t = parseNaturalTime(raw);
  if (!t) return { valid: false };
  if (t.hour < CLINIC_OPEN_HOUR || t.hour >= CLINIC_CLOSE_HOUR) return { valid: false };
  const ampm = t.hour >= 12 ? 'PM' : 'AM';
  const displayHour = t.hour % 12 || 12;
  const displayMinute = t.minute.toString().padStart(2, '0');
  return { valid: true, value: `${displayHour}:${displayMinute} ${ampm}` };
}

--- Main Turn Processing Dispatcher (server/src/lib/engine.ts:397-478) ---
export async function processConversationTurn(
  workflowId: string,
  userMessage: string,
  sessionId?: string
): Promise<{ sessionId: string; reply: string; replyPhraseKey?: string; toolCallExecuted?: string; savedCallId?: string }> {
  const session = await getOrCreateSession(workflowId, sessionId);
  session.currentLanguage = detectLanguageFromText(userMessage);
  const userMsgId = `msg_user_${Date.now()}`;
  session.clientMessages.push({ id: userMsgId, role: 'user', content: userMessage, timestamp: new Date().toISOString() });
  session.openAiMessages.push({ role: 'user', content: userMessage });
  const { openai, isConfigured, modelName } = getAIClient();
  let reply = '';
  let replyPhraseKey: string | undefined;
  let toolCallExecuted: string | undefined;
  const state = session.intakeState;

  if (state === 'completed') {
    reply = PHRASES.yes_confirm;
    replyPhraseKey = 'yes_confirm';
  } else if (state === 'awaiting_confirmation') {
    if (isConfirmationYes(userMessage)) {
      const booking = await bookAppointment(session);
      toolCallExecuted = 'create_calendar_event';
      session.intakeState = 'completed';
      session.isCompleted = true;
      reply = PHRASES.yes_confirm;
      replyPhraseKey = 'yes_confirm';
    } else if (isConfirmationNo(userMessage)) {
      session.intakeState = 'correcting';
      reply = PHRASES.no_prompt;
      replyPhraseKey = 'no_prompt';
    } else {
      reply = buildReadback(session.extractedFields);
    }
  } else if (state === 'correcting') {
    const target = detectCorrectionTarget(userMessage);
    if (target) {
      delete session.extractedFields[target];
      session.failCounts[target] = 0;
      const slot = INTAKE_SLOTS.find((s) => s.key === target)!;
      session.intakeState = slot.state;
      reply = slotPhrase(slot, false);
      replyPhraseKey = slot.question;
    } else {
      reply = PHRASES.no_prompt;
      replyPhraseKey = 'no_prompt';
    }
  } else {
    const slot = INTAKE_SLOTS.find((s) => s.state === state);
    if (!slot) {
      session.intakeState = 'collecting_name';
      reply = PHRASES.ask_name;
      replyPhraseKey = 'ask_name';
    } else {
      const candidate = isConfigured
        ? await extractSlotValueWithLLM(userMessage, slot, openai, modelName)
        : userMessage;
      const result = slot.validate(candidate);
      if (result.valid && result.value) {
        session.extractedFields[slot.key] = result.value;
        session.failCounts[slot.key] = 0;
        const currentIdx = INTAKE_SLOTS.findIndex((s) => s.key === slot.key);
        const nextSlot = INTAKE_SLOTS[currentIdx + 1];
        if (nextSlot) {
          session.intakeState = nextSlot.state;
          const ack = buildAcknowledgement(slot.key, result.value);
          reply = `${ack} ${slotPhrase(nextSlot, false)}`;
          replyPhraseKey = nextSlot.question;
        } else {
          session.intakeState = 'awaiting_confirmation';
          reply = buildReadback(session.extractedFields);
        }
      } else {
        session.failCounts[slot.key] = (session.failCounts[slot.key] ?? 0) + 1;
        const failures = session.failCounts[slot.key];
        if (failures >= MAX_FAILURES_PER_SLOT) {
          session.failCounts[slot.key] = 0;
          reply = PHRASES.offer_text;
          replyPhraseKey = 'offer_text';
        } else {
          const invalidKey = `sorry_invalid_${slot.key.replace('appt_', '')}` as keyof typeof PHRASES;
          reply = failures === 1 ? (PHRASES[invalidKey] || slotPhrase(slot, true)) : slotPhrase(slot, true);
          replyPhraseKey = slot.reask;
        }
      }
    }
  }
  return { sessionId: session.sessionId, reply, replyPhraseKey, toolCallExecuted };
}

4.1 State Machine & Hands-Free Loop
States:
- Initializing: Mount fetches session greeting.
- AI Assistant Speaking...: isPlayingAudio = true. Barge-in monitor listening.
- Listening — Turn Active: isRecording = true. Mic open, VAD checking for silence.
- Processing AI response...: isProcessing = true. Audio Blob sent to /api/conversations/:workflowId/voice.
- Call Session Completed: isCompleted = true. Session saved to Firestore.

Hands-Free Loop:
1. Assistant finishes speaking -> triggerAutoListen() triggers after 400ms delay.
2. startRecording() initializes mic stream, audio analyser, and MediaRecorder.
3. Audio meter calculates volume. When user stops speaking for VAD_SILENCE_MS (2000ms), stopRecording() is called.
4. submitTurn() sends audio blob to server.
5. Response arrives -> assistant speaks audio -> triggers step 1 again.

const speakBrowserTTS = (rawText: string, onAllDone?: () => void) => {
  try {
    if (!('speechSynthesis' in window)) {
      addVoiceLog(rawText, 'error', 'SpeechSynthesis API not supported in browser');
      if (onAllDone) onAllDone();
      else triggerAutoListen();
      return;
    }
    cancelAllSpeech();
    const normalized = normalizeForSpeech(rawText);
    const chunks = chunkTextForSpeech(normalized, 180);
    if (chunks.length === 0) {
      if (onAllDone) onAllDone();
      else triggerAutoListen();
      return;
    }
    speechQueueRef.current = [...chunks];
    const playNextChunk = () => {
      if (speechQueueRef.current.length === 0) {
        setIsPlayingAudio(false);
        isPlayingAudioRef.current = false;
        currentSpeakingTextRef.current = '';
        stopBargeInMonitor();
        if (onAllDone) onAllDone();
        else triggerAutoListen();
        return;
      }
      const chunk = speechQueueRef.current.shift()!;
      currentSpeakingTextRef.current = chunk;
      const utterance = new SpeechSynthesisUtterance(chunk);
      activeUtterancesRef.current.push(utterance); // Prevents Chrome GC bug
      const voice = availableVoices.find((v) => v.voiceURI === selectedVoiceURI);
      if (voice) {
        utterance.voice = voice;
        utterance.lang = voice.lang;

--- Barge-In Logic (client/src/pages/VoiceView.tsx:288-348) ---
const startBargeInMonitor = async () => {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({
      audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
    });
    bargeInStreamRef.current = stream;
    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    const source = ctx.createMediaStreamSource(stream);
    const analyser = ctx.createAnalyser();
    analyser.fftSize = 256;
    source.connect(analyser);
    bargeInAnalyserRef.current = analyser;
    const data = new Uint8Array(analyser.frequencyBinCount);
    const check = () => {
      if (!isPlayingAudioRef.current) { stopBargeInMonitor(); return; }
      analyser.getByteFrequencyData(data);
      let maxVal = 0;
      for (let i = 0; i < data.length; i++) if (data[i] > maxVal) maxVal = data[i];
      const vol = (maxVal / 255) * 100;
      if (vol >= BARGE_IN_SPEECH_THRESHOLD) {
        if (!bargeInStartTimeRef.current) bargeInStartTimeRef.current = Date.now();
        const speechDuration = Date.now() - bargeInStartTimeRef.current;
        if (speechDuration >= BARGE_IN_DETECT_MS) {
          handleBargeIn();
          return;
        }
      } else {
        bargeInStartTimeRef.current = null;
      }
      bargeInFrameRef.current = requestAnimationFrame(check);
    };
    bargeInFrameRef.current = requestAnimationFrame(check);
  } catch (err) {
    console.warn('[Barge-in] Could not start monitor:', err);
  }
};

const handleBargeIn = () => {
  cancelAllSpeech();
  abortControllerRef.current?.abort();
  setMessages((prev) => {
    const updated = [...prev];
    for (let i = updated.length - 1; i >= 0; i--) {
      if (updated[i].role === 'assistant') {
        updated[i] = { ...updated[i], content: updated[i].content + ' [interrupted]' };
        break;
      }
    }
    return updated;
  });
  stopBargeInMonitor();
  setIsPlayingAudio(false);
  isPlayingAudioRef.current = false;
  triggerAutoListen();
};

7. KNOWN ISSUES AND RISKS
================================================================================

1. Valid Names Being Rejected (>4 words):
   - Location: server/src/lib/validators.ts:37
   - Cause: words.length > 4 rejects names with more than 4 words. Long Indian names or names spoken with honorifics/prefixes (e.g. "Venkata Ramanathan Subramanian Iyer Sharma") fail validation and trigger a re-ask.

2. Single Initial Names Rejected if Stripped:
   - Location: server/src/lib/validators.ts:33
   - Cause: cleaned.length < 2 rejects single-letter inputs. While full names like "Ramesh Kumar S" succeed, single-letter surname answers fail.

3. Browser Echo at High Speaker Volumes:
   - Location: client/src/pages/VoiceView.tsx:320-330
   - Cause: If external laptop speakers are set to 100% volume without headphones, physical acoustic feedback into the mic can exceed BARGE_IN_SPEECH_THRESHOLD (50%) and interrupt the assistant mid-sentence.

4. SpeechSynthesis Pausing in Inactive Background Tabs:
   - Location: client/src/pages/VoiceView.tsx:944-946
   - Cause: Chromium engines throttle speechSynthesis and timers when the tab is blurred or backgrounded. A resume workaround (window.speechSynthesis.resume()) is present, but active tab focus is optimal.

5. Google Calendar Credentials Missing in Environment:
   - Location: server/src/lib/tools/calendar.ts:16-19
   - Cause: GOOGLE_CLIENT_ID and GOOGLE_CLIENT_SECRET are placeholders. The app gracefully falls back to Firestore-only booking, but calendar event creation is simulated.

6. Audio Always Sent to Server STT:
   - Location: client/src/pages/VoiceView.tsx:501
   - Cause: In continuous mode, the client always uploads the audio blob to /voice, consuming Sarvam STT quota on every turn even though the browser already computed a transcript via Web Speech API.
